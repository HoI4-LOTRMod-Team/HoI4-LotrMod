
increase_variag_economy = {
	if = {
		limit = {
			has_idea = KND_failing_economy
		}
		swap_ideas = {
			remove_idea = KND_failing_economy
			add_idea = KND_struggling_economy
		}
	}
	else_if = {
		limit = {
			has_idea = KND_struggling_economy
		}
		swap_ideas = {
			remove_idea = KND_struggling_economy
			add_idea = KND_weakened_economy
		}
	}
	else_if = {
		limit = {
			has_idea = KND_weakened_economy
		}
		swap_ideas = {
			remove_idea = KND_weakened_economy
			add_idea = KND_recovered_economy
		}
	}
	else_if = {
		limit = {
			has_idea = KND_recovered_economy
		}
		swap_ideas = {
			remove_idea = KND_recovered_economy
			add_idea = KND_strong_economy
		}
	}
	else_if = {
		limit = {
			has_idea = KND_strong_economy
		}
		swap_ideas = {
			remove_idea = KND_strong_economy
			add_idea = KND_prosperous_economy
		}
	}
}
