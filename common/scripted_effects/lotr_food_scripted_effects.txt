


@food_per_pop_factor = 0.017



lotr_food_init = {
	set_variable = { food_consumption_from_population = max_manpower_k }
	multiply_variable = { food_consumption_from_population = @food_per_pop_factor }
	multiply_variable = { food_consumption_from_population = food_consumption_from_population_factor }

	add_dynamic_modifier = {
		modifier = lotr_food_consumption_from_population
	}
}



lotr_food_update = {
	set_variable = { food_consumption_from_population = max_manpower_k }
	multiply_variable = { food_consumption_from_population = @food_per_pop_factor }
	multiply_variable = { food_consumption_from_population = modifier@food_use_from_population_factor }
}
