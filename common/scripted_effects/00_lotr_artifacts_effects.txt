lotr_artifacts_active_effects = {

	meta_effect = {
		text = {
			lotr_artifacts_effects__[ARTIFACT_ID] = yes
		}
		ARTIFACT_ID = "[?temp_selected_artifact.GetTokenKey]"
	}

	set_country_flag = {
		flag = artifact_cooldown_active
		value = 1
		days = 180
	}
	# TODO: after 180 days, lotr_artifacts_trigger_notification

}

lotr_artifacts_toggle_details_view = {
	if = {
		limit = { has_country_flag = ring_details_view_open }
		clr_country_flag = ring_details_view_open
	}
	if = {
		limit = { has_country_flag = lotr_artifacts_details_view_open }
		clr_country_flag = lotr_artifacts_details_view_open
	}
	# To keep the UI clean and smooth, we won't allow this to open when interception_form is open
	else_if = {
		limit = { NOT = { has_country_flag = ring_interception_form_open } }
		set_country_flag = lotr_artifacts_details_view_open
	}
}

lotr_transfer_all_artifacts = {
	# use special code to transfer ring
	if = {
		limit = {
			ring_has_the_ring = yes
		}
		PREV = { ring_seize_ring = yes }
	}

	while_loop_effect = {
		limit = {
			check_variable = { country_artifacts^num > 0 }
		}
		meta_effect = {
			text = {
				remove_ideas = [ARTIFACT_ID]
				PREV = {
					add_ideas = [ARTIFACT_ID]
				}
			}
			ARTIFACT_ID = "[?country_artifacts^0.GetTokenKey]"
		}
	}
}
