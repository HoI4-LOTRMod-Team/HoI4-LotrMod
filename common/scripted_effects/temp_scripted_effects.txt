gob_begin_overthrow_timer = {
    if = {
        limit = {
            gob_has_overthrow_timer = yes
        }

        set_variable = { factories_last_year = num_of_civilian_factories }
        random_list = {
            10 = {
                set_variable = { factories_demand_diff = 5 }
            }
            10 = {
                modifier = {
                    factor = 0.5
                    date < "3020.1.1"
                }
                set_variable = { factories_demand_diff = 7 }
            }
            10 = {
                set_variable = { factories_demand_diff = 4 }
            }
            10 = {
                modifier = {
                    factor = 0.5
                    date < "3021.1.1"
                }
                set_variable = { factories_demand_diff = 9 }
            }
        }
        set_variable = { factories_demand = factories_last_year }
        add_to_variable = { factories_demand = factories_demand_diff }

        activate_mission = GOB_overthrow_timer
    }
}
