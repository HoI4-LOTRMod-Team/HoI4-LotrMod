rgl_create_refugee_levy = {
    if = {
        limit = { NOT = { has_template = "Refugee Levy" } }
        division_template = {
            name = "Refugee Levy"
            regiments = {
                infantry = { x = 0 y = 0 }
                infantry = { x = 0 y = 1 }
                infantry = { x = 0 y = 2 }
                infantry = { x = 0 y = 3 }
            }
        }
    }
    create_unit = {
        division = "name = \"Refugee Levy\" division_template = \"Refugee Levy\" start_experience_factor = 0.0"
        count = 2
        allow_spawning_on_enemy_provs = no
        owner = RGL
    }
}
