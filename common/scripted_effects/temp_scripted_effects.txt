    if = {
        limit = {
            gob_has_overthrow_timer = yes
        }

        random_list = {
            10 = {
                set_variable = { factories_demand_diff = 5 }
            }
            10 = {
                modifier = {
                    factor = 0.5
                    date < "3020.1.1"
                }
                set_variable = { factories_demand_diff = 7 }
            }
            10 = {
                set_variable = { factories_demand_diff = 4 }
            }
            10 = {
                modifier = {
                    factor = 0.5
                    date < "3021.1.1"
                }
                set_variable = { factories_demand_diff = 9 }
            }
        }
        set_variable = { factories_demand = factories_last_year }
        add_to_variable = { factories_demand = factories_demand_diff }

        activate_mission = GOB_overthrow_timer
    }
}

do_rhun_civil_war = {
    
    # TODO: News event

    # transfer states
    for_each_loop = {
        array = RHU_list_states_civil_war
        DRH = {
            transfer_state = v
        }
    }

    # Set politics
    RHU = {
        set_politics = {
            ruling_party = belligerent
            elections_allowed = no
        }

        add_popularity = {
            ideology = cooperative
            popularity = -1.0
        }
        add_popularity = {
            ideology = unaligned
            popularity = -0.2
        }
        add_popularity = {
            ideology = revolutionary
            popularity = -1.0
        }
    }
    DRH = {
        set_politics = {
            ruling_party = cooperative
            elections_allowed = no
        }

        set_popularities = {
            cooperative = 61
            unaligned = 24
            belligerent = 13
            revolutionary = 2
        }
    }

    # magical equipment infusion
    RHU = {
        add_equipment_to_stockpile = {
            type = infantry_equipment_0
            amount = 3000
        }
        add_timed_idea = {
            idea = RHUN_civil_war
