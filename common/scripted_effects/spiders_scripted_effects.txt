




spiders_country_setup = {
	SPI = {
		set_variable = { casualties_inflicted_to_mp_factor = 0.5 }
		set_variable = { casualties_taken_to_mp_factor = 0.25 }

		set_variable = { mp_gained_from_enemy_casualties = 0 } # in thousands
		set_variable = { mp_gained_from_own_casualties = 0 } # in thousands
		set_variable = { mp_gained_total = 0 } # in thousands

		set_variable = { taken_casualties_prev = 0 }

		init_casualties_counts = yes
	}
}



spiders_weekly_tick = {
	SPI = {

		# NOTE: as cursed as the count_casualties code is, I've run it through the profiler and it seems to run remarkably fast (literally <1ms), so fuck it
		count_casualties_inflicted = yes

		if = {
			limit = {
				check_variable = { casualties_this_week > 0 }
			}

			if = {
				limit = {
					has_completed_focus = SPI_tender_flesh # the focus that gives more for elves
					check_variable = { casualties_on_elves_this_week > 0 }
				}
				# add 0.5 times this to this week's casualties
				multiply_variable = { casualties_on_elves_this_week = 0.5 }
				add_to_variable = { casualties_this_week = casualties_on_elves_this_week }
			}

			if = {
				limit = {
					has_completed_focus = SPI_black_blood # the focus that gives more for orcs
					check_variable = { casualties_on_orcs_this_week > 0 }
				}
				# add 0.5 times this to this week's casualties
				multiply_variable = { casualties_on_orcs_this_week = 0.5 }
				add_to_variable = { casualties_this_week = casualties_on_orcs_this_week }
			}

			# calculate casualties diff
			set_temp_variable = { temp_casualties_diff = casualties_this_week }
			multiply_temp_variable = { temp_casualties_diff = casualties_inflicted_to_mp_factor }

			# add equivalent manpower
			add_to_variable = { mp_gained_from_enemy_casualties = temp_casualties_diff }
			multiply_temp_variable = { temp_casualties_diff = 1000 }
			add_manpower = temp_casualties_diff
		}

		if = {
			limit = {
				check_variable = { casualties_k > taken_casualties_prev }
			}
			set_temp_variable = { temp_casualties_diff = casualties_k }
			subtract_from_temp_variable = { temp_casualties_diff = taken_casualties_prev }
			multiply_temp_variable = { temp_casualties_diff = casualties_taken_to_mp_factor }

			# add equivalent manpower
			add_to_variable = { mp_gained_from_own_casualties = temp_casualties_diff }
			multiply_temp_variable = { temp_casualties_diff = 1000 }
			add_manpower = temp_casualties_diff

			# set prev
			set_variable = { taken_casualties_prev = casualties_k }
		}

		set_variable = { mp_gained_total = 0 }
		add_to_variable = { mp_gained_total = mp_gained_from_own_casualties }
		add_to_variable = { mp_gained_total = mp_gained_from_enemy_casualties }
	}
}

# states bordering the Mirkwood (wood, not country):
# 193 90 211 94 212 96 99 97 205 227 191 198 261 199 71 196 260
# NOTE: If the Mirkwood borders change, this needs to be adjusted!
# or, alternatively, find a way to compute it automatically
spiders_daily_tick = {
	if = {
		limit = {
			has_completed_focus = SPI_the_mothers_guard
		}


		if = {
			limit = {
				NOT = { has_completed_focus = SPI_extend_the_perimeter }
				num_battalions_in_states = {
					count > 0
					states = { 229 71 260 193 226 83 80 } # states bordering Spider's nest state
					types = { nest_mothers }
				}
			}

			country_event = spiders.105
		}

		else_if = {
			limit = {
				num_battalions_in_states = {
					count > 0
					states = { 193 90 211 94 212 96 99 97 205 227 191 198 261 199 71 196 260 } # states bordering all of mirkwood
					types = { nest_mothers }
				}
			}

			country_event = spiders.105
		}
	}

	set_variable = { global.days_until_3021 = 1103030 }
	subtract_from_variable = { global.days_until_3022 = global.num_days }
	clamp_variable = { var = global.days_until_3022 min = 0 }
}
