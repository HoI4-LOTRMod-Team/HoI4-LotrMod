




spiders_country_setup = {
	SPI = {
		set_variable = { inflicted_casualties = 0 }
		set_variable = { inflicted_casualties_prev = 0 }

		set_variable = { taken_casualties_prev = 0 }

		set_variable = { casualties_inflicted_to_mp_factor = 1000.0 }
		set_variable = { casualties_taken_to_mp_factor = 1.0 }

		set_variable = { mp_gained_from_enemy_casualties = 0 }
		set_variable = { mp_gained_from_own_casualties = 0 }

		init_casualties_counts = yes
	}
}




spiders_weekly_tick = {
	SPI = {

		# NOTE: as cursed as the count_casualties code is, I've run it through the profiler and it seems to run remarkably fast (literally just 1ms), so fuck it
		count_inflicted_casualties_on_current_enemies = yes


		# NOTE AND TODO:
		# I've replace the thing above with the "current_enemies" as opposed to "all_countries", because it seems to actually work (unlike the other thing)
		# This *could* lead to a situation where the spiders deal a *shit* ton of casualties, peace out on the same week, and then don't get the manpower. Do we care? ... Probably not...

		# Secondly: These casualties (inflicted_casualties and everything else) is counted in *thousands*. I've set the "casualties_inflicted_to_mp_factor" to 1k because of that, but idk maybe that shouldnt be there...
		# Thirdly: The "from own casualties" remains kind of untested. I recommend looking at that again
		# Fourthly: It only update in *full* thousands. Meaning you need 1001 casualties inflicted before it registers that at all. It will see 999 as 0.
		#				as such, it *might* be an idea to use a wording of "for every 1000 casualties we inflict, receive 500 manpower" etc.

		if = {
			limit = {
				check_variable = { inflicted_casualties > inflicted_casualties_prev }
			}

			# calculate casualties diff
			set_temp_variable = { temp_casualties_diff = inflicted_casualties }
			subtract_from_temp_variable = { temp_casualties_diff = inflicted_casualties_prev }
			multiply_temp_variable = { temp_casualties_diff = casualties_inflicted_to_mp_factor }

			# add equivalent manpower
			add_manpower = temp_casualties_diff
			add_to_variable = { mp_gained_from_enemy_casualties = temp_casualties_diff }

			# update prev
			set_variable = { taken_casualties_prev = inflicted_casualties }
		}

		if = {
			limit = {
				check_variable = { inflicted_casualties_prev > taken_casualties_prev }
			}

			# calculate casualties diff
			set_temp_variable = { temp_casualties_diff = casualties }
			subtract_from_temp_variable = { temp_casualties_diff = taken_casualties_prev }
			multiply_temp_variable = { temp_casualties_diff = casualties_taken_to_mp_factor }

			# add equivalent manpower
			add_manpower = temp_casualties_diff
			add_to_variable = { mp_gained_from_own_casualties = temp_casualties_diff }

			# update prev
			set_variable = { taken_casualties_prev = casualties }
		}
	}
}

# 193 90 211 94 212 96 99 97 205 227 191 198 261 199 71 196 260
# NOTE: If the Mirkwood borders change, this needs to be adjusted!
# or, alternatively, find a way to compute it automatically
spiders_daily_tick = {
	if = {
		limit = {
			has_completed_focus = SPI_the_mothers_guard
			NOT = { has_completed_focus = SPI_extend_the_perimeter }
		}


		if = {
			limit = {
				num_battalions_in_states = {
					count > 0
					states = { 229 71 260 193 226 83 80 } # states bordering Spider's nest state
					types = { nest_mothers }
				}
			}

			country_event = spiders.105
		}
	}
	else_if = {
		limit = {
			has_completed_focus = SPI_the_mothers_guard
			has_completed_focus = SPI_extend_the_perimeter
		}

		if = {
			limit = {
				num_battalions_in_states = {
					count > 0
					states = { 193 90 211 94 212 96 99 97 205 227 191 198 261 199 71 196 260 } # states bordering all of mirkwood
					types = { nest_mothers }
				}
			}

			country_event = spiders.105
		}
	}
}
