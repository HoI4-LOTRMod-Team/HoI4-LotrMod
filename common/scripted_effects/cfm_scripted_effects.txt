# disable the cfm mechanic for this country (only allows SPI or DGU. Disabling for MIR requires both SPI and DGU to be off)
cfm_deactivate_for_country = {
	clr_country_flag = cfm_active
}


cfm_increase_5p = {
	if = {
		limit = { has_cfm_active = yes }
		add_to_variable = {
			var = cfm_presence
			value = 0.05
			tooltip = "TODO"
		}
	}
}

cfm_weekly_update = {
	every_country = {
		limit = { has_cfm_active = yes }
		cfm_update_presence_weekly = yes
	}
}

cfm_update_presence_weekly = {
	set_temp_variable = { temp_presence_gain = modifier@cfm_presence_gain }
	set_temp_variable = { temp_presence_gain_mult = modifier@cfm_presence_gain_factor }
	add_to_temp_variable = { temp_presence_gain_mult = 1.0 }
	multiply_temp_variable = { temp_presence_gain = temp_presence_gain_mult }

	# increase presence
	add_to_variable = { cfm_presence = temp_presence_gain }
}

cfm_reset_presence = {
	set_variable = {
		cfm_presence = 0
	}
}
