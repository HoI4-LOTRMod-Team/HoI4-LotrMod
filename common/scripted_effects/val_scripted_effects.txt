        limit = { has_focus_tree = rhosgobel }

                has_country_flag = RGL_the_elven_kings_shield_taken

            uncomplete_national_focus = {
                focus = RGL_the_elven_kings_shield
                refund_political_power = no

rgl_create_refugee_levy = {
    if = {
        limit = { NOT = { has_template = "Refugee Levy" } }
        division_template = {
            name = "Refugee Levy"
            regiments = {
                infantry = { x = 0 y = 0 }
                infantry = { x = 0 y = 1 }
                infantry = { x = 0 y = 2 }
                infantry = { x = 0 y = 3 }
            }
            priority = 0
        }
    }
    random_controlled_state = {
        create_unit = {
            division = "name = \"Refugee Levy\" division_template = \"Refugee Levy\" start_experience_factor = 0.0"
            count = 4
            allow_spawning_on_enemy_provs = no
            owner = RGL
        }
    }
}
