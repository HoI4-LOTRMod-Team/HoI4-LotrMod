




deploy_mirkwood_hidden_guard = {
	set_country_flag = mirkwood_hidden_guard_deployed
	country_event = mirkwood.15
}


mir_evacuate_state_industry = {
    if = {
        limit = { arms_factory > 0 }
        set_temp_variable = { temp_factories = arms_factory_level }
        add_to_variable = { MIR.mir_evacuated_mils = temp_factories }
        custom_effect_tooltip = MIR_evacuate_industry_mils_tt
        hidden_effect = {
            remove_building = { type = arms_factory level = temp_factories }
        }
    }

    if = {
        limit = { industrial_complex > 0 }
        set_temp_variable = { temp_factories = industrial_complex_level }
        add_to_variable = { MIR.mir_evacuated_civs = temp_factories }
        custom_effect_tooltip = MIR_evacuate_industry_civs_tt
        hidden_effect = {
            remove_building = { type = industrial_complex level = temp_factories }
        }
    }
}

mir_evacuate_state_population = {
    set_temp_variable = { temp_population_amount = state_population_k }
    multiply_temp_variable = { temp_population_amount = 0.5 }
    add_to_variable = { MIR.mir_evacuated_population_k = temp_population_amount }
    custom_effect_tooltip = MIR_evacuate_population_evac_tt
    multiply_temp_variable = { temp_population_amount = -1000 }
    hidden_effect = {
        add_manpower = temp_population_amount
    }
}
