cfm_increase_presence_in_state = {
	if = {
		limit = { has_state_flag = cfm_enabled }

		if = {
			limit = { PREV = { tag = MIR } }
			add_to_variable = { cfm_presence_MIR = 1 }
		}
		else_if = {
			limit = { PREV = { tag = DGU } }
			add_to_variable = { cfm_presence_DGU = 1 }
		}
		else_if = {
			limit = { PREV = { tag = SPI } }
			add_to_variable = { cfm_presence_SPI = 1 }
		}
		else = {
			log = "Something went wrong!"
		}
		cfm_on_state_presence_update = yes
	}

	# TODO: tooltip
}



cfm_on_state_presence_update = {
	# NOTE/TODO: maybe instead of transfer state it should start a mission/decision to let the player have time to react
	if = {
		limit = {
			NOT = { is_controlled_by = MIR }
			check_variable = { cfm_presence_MIR > var:cfm_presence_DGU }
			check_variable = { cfm_presence_MIR > var:cfm_presence_SPI }
		}
		transfer_state_to = MIR
	}
	else_if = {
		limit = {
			NOT = { is_controlled_by = DGU }
			check_variable = { cfm_presence_MIR > var:cfm_presence_MIR }
			check_variable = { cfm_presence_MIR > var:cfm_presence_SPI }
		}
		transfer_state_to = DGU
	}
	else_if = {
		limit = {
			NOT = { is_controlled_by = SPI }
			check_variable = { cfm_presence_MIR > var:cfm_presence_DGU }
			check_variable = { cfm_presence_MIR > var:cfm_presence_MIR }
		}
		transfer_state_to = SPI
	}
}
