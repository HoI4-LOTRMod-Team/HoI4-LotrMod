	GOB_overthrow_timer = {
		icon = generic_research

		fire_only_once = no # allow decision to reappear
		days_mission_timeout = 365
		selectable_mission = no
		is_good = no # is the timer running out a good thing?

		allowed = {
			always = no
		}

		activation = {
			always = no
		}

		available = {
			gob_has_not_overthrow_timer = yes
		}

		timeout_effect = {
			# when the timer runs out uninterrupted
			# TODO: This should 100% contain an event
			hidden_effect = {
				add_popularity = {
					ideology = revolutionary
					popularity = 0.1
				}
				kill_country_leader = yes
				hidden_effect = {
					GOB_no_leader = {
						add_country_leader_role = {
							promote_leader = yes
							country_leader = {
								ideology = mutinous
								traits = { gob_power_vacuum }
							}
						}
						add_country_leader_role = {
							promote_leader = yes
							country_leader = {
								ideology = secluded
								traits = { gob_power_vacuum }
							}
						}
						add_country_leader_role = {
							promote_leader = yes
							country_leader = {
								ideology = federative
								traits = { gob_power_vacuum }
							}
						}
						add_country_leader_role = {
							promote_leader = yes
							country_leader = {
								ideology = expansionist
								traits = { gob_power_vacuum }
							}
						}
					}
				}
				add_timed_idea = {
					idea = GOB_power_vacuum
					days = 90
				}
			}
			custom_effect_tooltip = gob_overthrow_timer_tt
		}
	}
}
