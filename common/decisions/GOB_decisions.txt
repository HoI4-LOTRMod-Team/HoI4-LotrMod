						add_country_leader_role = {
							promote_leader = yes
							country_leader = {
								ideology = mutinous
								traits = { gob_power_vacuum }
							}
						}
						add_country_leader_role = {
							promote_leader = yes
							country_leader = {
								ideology = secluded
								traits = { gob_power_vacuum }
							}
						}
						add_country_leader_role = {
							promote_leader = yes
							country_leader = {
								ideology = federative
								traits = { gob_power_vacuum }
							}
						}
						add_country_leader_role = {
							promote_leader = yes
							country_leader = {
								ideology = expansionist
								traits = { gob_power_vacuum }
							}
						}
					}
				}
				add_timed_idea = {
					idea = GOB_power_vacuum
					days = 90
				}
			}
			custom_effect_tooltip = gob_overthrow_timer_tt
		}
	}
