            }

	MIR_withdraw_southern_troops = {
		icon = GFX_decision_generic_operation

		fire_only_once = yes
		cost = 0

		allowed = {
			tag = MIR
		}

		visible = {
			83 = { is_fully_controlled_by = MIR }
			OR = {
				190 = {
					NOT = { is_controlled_by = MIR }
					controller = { NOT = { is_ally_with = MIR } NOT = { has_war_with = MIR } }
				}
				262 = {
					NOT = { is_controlled_by = MIR }
					controller = { NOT = { is_ally_with = MIR } NOT = { has_war_with = MIR } }
				}
			}
		}

		available = {
			always = yes
		}

		ai_will_do = {
			factor = 100
		}
		
		complete_effect = {
			every_state = {
				limit = {
					is_fully_controlled_by = MIR
					has_state_flag = cfm_south_of_narrows
				}
				teleport_armies = {
					limit = { tag = MIR }
					to_state = 83
				}
			}
		}
