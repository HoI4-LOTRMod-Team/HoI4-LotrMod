on_actions = {

	on_war_relation_added = {
		effect = {
			# DGU declares war on LTH with DGU_focus_on_lothlorien focus
			if = {
				limit = { ROOT = { original_tag = DGU has_completed_focus = DGU_focus_on_lothlorien } FROM = { original_tag = LTH } }

				ROOT = {
					add_timed_idea = {
						idea = DGU_focus_on_lothlorien
						days = 140
					}
				}
			}

			# DGU declares war on MIR with DGU_focus_on_mirkwood focus
			if = {
				limit = { ROOT = { original_tag = DGU has_completed_focus = DGU_focus_on_mirkwood } FROM = { original_tag = MIR } }

				ROOT = {
					add_timed_idea = {
						idea = DGU_focus_on_mirkwood
						days = 140
					}
				}
			}
		}
	}

	on_weekly_DGU = {
		effect = {
			if = {
				limit = {
					has_completed_focus = DGU_the_masters_of_the_mirkwood
					NOT = { has_country_flag = dgu_lost_mirkwood_control }
					control_all_mirkwood_states = no
				}
				set_country_flag = dgu_lost_mirkwood_control
			}
		}
	}

	on_weekly_MIR = {
		effect = {
			if = {
				limit = {
					has_completed_focus = MIR_dominion_over_the_mirkwood
					NOT = { has_country_flag = dgu_lost_mirkwood_control }
					control_all_mirkwood_states = no
				}
				set_country_flag = dgu_lost_mirkwood_control
			}
		}
	}

	on_weekly_SPI = {
		effect = {
			if = {
				limit = {
					has_completed_focus = SPI_purge_the_two_leggeds
					NOT = { has_country_flag = dgu_lost_mirkwood_control }
					control_all_mirkwood_states = no
				}
				set_country_flag = dgu_lost_mirkwood_control
			}
		}
	}
}
