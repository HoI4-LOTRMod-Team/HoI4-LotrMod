

on_actions = {
	
	on_civil_war_end = {
        effect = {
            if = {
				limit = {
					ROOT = { tag = DRH has_civil_war = yes }
					FROM = { tag = RHU }
				}
				DRH = { complete_national_focus = DRH_treaty_of_mistrand }
			}
		}
	}

	
	on_weekly_DRH = {
		effect = {
			if = {
				limit = {
					has_civil_war = yes
					country_exists = RHU
					surrender_progress > 0.7
				}
				DRH = { country_event = rhun.116 }
			}
		}
	}
}
