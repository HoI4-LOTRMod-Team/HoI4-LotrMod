on_actions = {

	# New general -> assign race trait
	on_unit_leader_promote_from_ranks_veteran = {
		effect = {
			# Mirkwood
			if = {
				limit = { OWNER = { original_tag = MIR has_completed_focus = MIR_call_for_lineage_records NOT = { has_completed_focus = MIR_martial_trials } } }
				ROOT = {
					random_list = {
						50 = {
							add_unit_leader_trait = highborn_elf
						}
						50 = {
							modifier = {
								factor = 0
								OWNER = { has_completed_focus = MIR_restrict_command_to_highborn_captrains }
							}
							
						}
					}
				}
			}
			else_if = {
				limit = { OWNER = { original_tag = MIR has_completed_focus = MIR_martial_trials } }
				ROOT = {
					add_unit_leader_trait = highborn_elf_2
				}
			}

			# Spiders
			if = {
				limit = { OWNER = { original_tag = SPI NOT = { has_completed_focus = SPI_exalt_the_brood_warriors } } }
				ROOT = {
					add_unit_leader_trait = instinct_driven
				}
			}
			else_if = {
				limit = { OWNER = { original_tag = SPI has_completed_focus = SPI_exalt_the_brood_warriors } }
				ROOT = {
					add_unit_leader_trait = instinct_driven_improved
				}
			}
		}
	}

	# Spiders weekly tick
	on_weekly_SPI  = {

		effect = {
			spiders_weekly_tick = yes
		}
	}

	on_daily_SPI = {

		effect = {
			spiders_daily_tick = yes
		}

	}

	on_monthly_MIR = {
		effect = {
			if = {
				limit = { has_country_flag = mir_spider_growth_active }
				add_unit_bonus = {
					mirstory_spider_infantry = {
						soft_attack = 0.005
						hard_attack = 0.005
						defense = 0.005
						breakthrough = 0.005
					}
				}
			}
		}
	}

	on_daily_MIR = {
		effect = {
			cfm_weekly_update = yes

			if = {
				limit = {
					has_completed_focus = MIR_the_kings_hidden_guard
					NOT = { has_country_flag = mirkwood_hidden_guard_deployed }
					any_enemy_country = {
						divisions_in_state = {
    						size > 0
    						state = 83
						}
					}
				}
				deploy_mirkwood_hidden_guard = yes
			}
		}
	}
