


on_actions = {

	on_startup = {
		effect = {
			GOB = {
				set_variable = { scrap = 0 }
				set_variable = { equipment_produced = 0 }
			}
		}
	}

	on_weekly_GOB = {
		effect = {

			if = {
				limit = { has_completed_focus = GOB_nothing_goes_to_waste }

				set_temp_variable = { temp_equip_diff = total_equipment_produced_infantry }
				subtract_from_temp_variable = { temp_equip_diff = equipment_produced }
				multiply_variable = { temp_equip_diff = 0.1 }
				add_to_variable = { bonus_scrap_amount = temp_equip_diff }
			}
		}
	}

	on_monthly_GOB = {
		effect = {
			if = {
				limit = {
					has_variable = gob_yearly_mcounter
					check_variable = { gob_yearly_mcounter < 12 }
				}
				add_to_variable = { gob_yearly_mcounter = 1 }
			}
			else = {
				set_variable = { gob_yearly_mcounter = 1 }
				gob_on_yearly = yes
			}
		}
			
	}
}
