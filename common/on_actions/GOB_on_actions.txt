


on_actions = {

	on_startup = {
		effect = {
			GOB = {
				set_variable = { scrap = 0 }
				set_variable = { equipment_produced = 0 }
			}
		}
	}

	on_weekly_GOB = {
		effect = {

			if = {
				limit = { has_completed_focus = GOB_nothing_goes_to_waste }

				set_temp_variable = { temp_equip_diff = total_equipment_produced_infantry }
				subtract_from_temp_variable = { temp_equip_diff = equipment_produced }
				multiply_variable = { temp_equip_diff = 0.1 }
				add_to_variable = { bonus_scrap_amount = temp_equip_diff }
			}
		}
	}

	on_monthly_GOB = {
		effect = {
			if = {
				limit = {
					has_variable = gob_yearly_mcounter
					check_variable = { gob_yearly_mcounter < 12 }
				}
				add_to_variable = { gob_yearly_mcounter = 1 }
			}
			else = {
				set_variable = { gob_yearly_mcounter = 1 }
				gob_on_yearly = yes
			}
		}
	}

	on_capitulation = {
    	effect = {
			if = {
				limit = { ROOT = { has_country_flag = gob_country_has_mercenaries } GOB = { exists = yes } }
				ROOT = {
					gob_remove_mercs_and_template = yes
				}
				GOB = {
					gob_create_merc_divisions_equipped = yes
					country_event = goblintown.116
				}
			}
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					ROOT = { has_country_flag = gob_country_has_mercenaries }
					GOB = { exists = yes has_completed_focus = GOB_war_loot }
					NOT = { FROM.FROM = { has_state_flag = gob_state_looted } }
				}
				FROM.FROM = { set_state_flag = gob_state_looted }
				GOB = { add_to_variable = { scrap = 800 } }
			}
		}
	}
}
