

# Context: The state files theoretically have a feature letting us do this "force_link_ownership_to = 206".
# But as of right now, it does *not* work. This is why we do it manually in here.

on_actions = {
	

	on_state_control_changed = {
		effect = {
			
			every_state = {
				limit = {
					impassable = yes
					has_variable = force_linked_to
					check_variable = { force_linked_to = FROM.FROM.id }
				}
				transfer_state_to = ROOT
			}

		}
	}


}
