on_actions = {

	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					ROOT = {
						tag = GLD
						has_completed_focus = GLD_war_loans
						NOT = { has_war_with = GOB }
						GOB = {
							has_country_flag = war_loan_active
						}
					}
					FROM.FROM = {
						industrial_complex > 0
					}
				}
				FROM.FROM = {
					set_temp_variable = { factory_amount = industrial_complex }
					if = {
						limit = { check_variable = { factory_amount < GOB.war_loan_amount } }
						set_temp_variable = { factory_amount = GOB.war_loan_amount }
					}
					remove_building = { type = industrial_complex level = factory_amount }
					GOB = {
						random_core_state = {
							add_building_construction = {
								type = industrial_complex
								level = 1
								instant_build = yes
							}
						}
					}
					subtract_from_variable = { GOB.war_loan_amount = factory_amount }
					if = {
						limit = { check_variable = { GOB.war_loan_amount < 1 } }
						GOB = { clr_country_flag = war_loan_active }
					}
				}
				
			}
		}
	}
}
