


on_actions = {

	on_daily_MOR = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = morm_disabled } }
				morm_recalculate_menace = yes
			}
		}
	}

	on_weekly_MOR = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = morm_disabled } }
				morm_weekly = yes
			}
		}
	}

	on_capitulation_immediate = {
		effect = {
			if = {
				limit = { ROOT = { tag = MOR has_government = belligerent } }
				morm_end_menace = yes

				if = {
					limit = {
						ROOT = { has_country_leader = { character = MOR_sauron_eye ruling_only = yes } }
					}
					ROOT = {
						MOR_sauron_eye = {
							lotr_kill_character = yes
						}
					}
				}
			}
		}
	}
	
}
