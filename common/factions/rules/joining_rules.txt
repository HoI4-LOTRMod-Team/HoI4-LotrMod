joining_rule_non_belligerent = {
	type = joining_rules
	visible = {
		NOT = { has_government = belligerent }
	}
	ai_will_do = {
		base = 0
		modifier = {
			add = 1
			NOT = { has_government = belligerent }
		}
		modifier = {
			add = 1
			has_government = revolutionary
		}
	}
	trigger = {
		NOT = {
			has_government = belligerent
		}
	}
}

joining_rule_non_revolutionary = {
	type = joining_rules
	visible = {
		NOT = { has_government = revolutionary }
	}
	ai_will_do = {
		base = 0
		modifier = {
			add = 1
			NOT = { has_government = revolutionary }
		}
		modifier = {
			add = 1
			has_government = belligerent
		}
	}
	available = {
		IF = {
			limit = {
				has_faction_template = faction_template_allies
			}
			NOT = {
				any_enemy_country = {
					is_major = yes
					has_government = belligerent
				}
			}
		}
	}
	can_remove = {
		always = yes 
		
	}
	trigger = {
		NOT = {
			has_government = revolutionary 
		}
	}
}

		}
	}
	
	trigger = {
		any_neighbor_country = {
			is_in_faction_with = FROM
		}
	}
	modifier = {
		enemy_justify_war_goal_time = 0.2
	}
}
