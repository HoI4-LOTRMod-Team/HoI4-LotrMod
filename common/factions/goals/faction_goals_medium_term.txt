faction_goal_heavy_infantry = {
	name = heavy_infantry_goal_name
	description = heavy_infantry_goal_desc
	category = medium_term
	group = FOCUS_FILTER_ARMY_XP
	available = {
		all_allied_country = {
			custom_trigger_tooltip = {
				tooltip = custom_tooltip_faction_goal_has_6_or_less_heavy_infantry
				check_variable = { var:num_armies_with_type@heavy_infantry < 6 }
			}
		}
	}
	completed = {
		all_allied_country = {
			custom_trigger_tooltip = {
				tooltip = custom_tooltip_faction_goal_has_atleast_6_heavy_infantry
				check_variable = { var:num_armies_with_type@heavy_infantry > 6 }
			}
		}
	}
	complete_effect = {
		add_faction_power_projection = 100
		custom_effect_tooltip = generic_skip_one_line_tt
		every_collection_element = {
			input = {
				input = game:scope
				operators = { 
					faction_members 
					limit = {
						check_variable = { var:num_armies_with_type@heavy_infantry > 6 }
					}
				}
				name = custom_tooltip_faction_goal_all_faction_members_with_atleast_1_heavy_infantry
			}
			random_character = {
				tooltip = ONE_GENERAL	
				visible_when_empty = yes
				limit = {
					is_army_leader = yes
					is_field_marshal = no
					NOT = { has_trait = cavalry_officer }
					NOT = { has_trait = panzer_leader }
					NOT = { has_trait = invader }
				}
				gain_xp = 400
				add_attack = 1
				add_trait = { trait = invader }
			}
		}
		custom_effect_tooltip = generic_skip_one_line_tt
		custom_effect_tooltip = ALL_FACTION_MEMBERS
		every_faction_member = {
			add_tech_bonus = {
				name = marine_corps_tech_bonus
				bonus = 0.75
				uses = 1
				category = lotr_infantry_specialforces
			}
		}
	}
	ai_will_do = {
		base = 0
		modifier = {
			add = 1
			any_state = {
    		    is_controlled_by_ROOT_or_ally = yes
    		    is_coastal = yes
    		}
		}
		modifier = {
			add = 2
			any_state = {
    		    is_controlled_by = ROOT
    		    is_coastal = yes
    		}
		}
	}
}
