





# @docs(Generic): Starting a war between PREV and THIS is possible (no NA pacts, allies or existing wars)
war_with_is_possible = {
    custom_trigger_tooltip = {
        tooltip = war_with_is_possible_tt
        exists = yes
        NOT = { is_ally_with = PREV }
        NOT = { has_war_with = PREV }
        NOT = { has_non_aggression_pact_with = PREV }
        NOT = { has_truce_with = PREV }   
    }
}

# @docs(Generic): ROOT can invite THIS to a faction
can_invite_to_faction = {
    custom_trigger_tooltip = {
        tooltip = can_invite_to_faction_tt
        ROOT = {
            is_subject = no
            is_faction_leader = yes
            NOT = { eternal_animosity_exists_between = yes }
        }
        exists = yes
        is_subject = no
        is_in_faction = no
        NOT = { has_war_with = ROOT }
    }
}

# @docs(Generic): ROOT can ask THIS to join their faction
can_ask_to_join_faction = {
    custom_trigger_tooltip = {
        tooltip = can_ask_to_join_faction_tt
        ROOT = {
            is_subject = no
            NOT = { has_war_with = PREV }    
            is_in_faction = no
            exists = yes
        }
        exists = yes
        is_subject = no
        is_faction_leader = yes
        NOT = { eternal_animosity_exists_between = yes }
    }
}


# @docs(Generic): Will declaring war cause trouble with other countries?
is_isolated_target = {
    is_in_faction = no
    NOT = { any_country = { has_guaranteed = PREV } }
    is_subject = no
}

# @docs(Generic): ROOT or one of their allies is a neighbor of THIS
can_reach = {
    custom_trigger_tooltip = {
        tooltip = can_reach_tt
        exists = yes
        OR = {
            is_neighbor_of = ROOT
            any_neighbor_country = {
                is_ally_with = ROOT
            }
        }
    }
}

# @docs(Generic): THIS is behind on infantry equipment research
is_behind_on_inf_equipment = {
    OR = {
        AND = {
            NOT = { has_tech = infantry_weapons2 }
            NOT = { has_tech = cheap_infantry_weapons2 }
            date > 3018.5.1
        }
        AND = {
            NOT = { has_tech = improved_infantry_weapons }
            NOT = { has_tech = cheap_improved_infantry_weapons }
            date > 3020.1.1
        }
        AND = {
            NOT = { has_tech = advanced_infantry_weapons }
            NOT = { has_tech = cheap_advanced_infantry_weapons }
            date > 3023.1.1
        }
        AND = {
            NOT = { has_tech = universal_infantry_weapons }
            date > 3025.1.1
        }
    }
}


# @docs(Generic): all of THIS' core states are controlled by ROOT or their subjects
all_core_states_controlled_or_puppeted = {
    custom_trigger_tooltip = {
        tooltip = all_core_states_controlled_or_puppeted_tt
        all_core_state = {
            OR = {
                is_controlled_by = ROOT
                controller = { is_subject_of = ROOT }
            }
        }
    }
}

# Note: This is a rough approximate to calculate how many civs we get from trade by taking the total, any additionals or offmaps as well as factories in states
# There is a lot of stuff where this approximation could be wrong (non-core, other offmap sources etc.)
gob_factories_from_trade_limit = {
    
    set_temp_variable = { temp_civs = num_of_civilian_factories }

    subtract_from_temp_variable = { temp_civs = modifier@extra_civilian_factories }
    
    all_controlled_state = {
        subtract_from_temp_variable = { PREV.temp_civs = building_level@industrial_complex }
        always = yes
    }

    subtract_from_temp_variable = { temp_civs = interest_factories }

    check_variable = {
        temp_civs > 2
    }

}
