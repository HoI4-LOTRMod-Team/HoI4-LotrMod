



# Either one of the other states is an ally, or we control all their core states
is_vales_confederation_member = {
    if = {
        limit = { tag = VAL }
        OR = {
            is_ally_with = RGL
            is_ally_with = GLD
            RGL = { all_core_state = { is_controlled_by = VAL } }
            GLD = { all_core_state = { is_controlled_by = VAL } }
        }
    }
    else_if = {
        limit = { tag = GLD }
        OR = {
            is_ally_with = RGL
            is_ally_with = VAL
            RGL = { all_core_state = { is_controlled_by = GLD } }
            VAL = { all_core_state = { is_controlled_by = GLD } }
        }
    }
    else_if = {
        limit = { tag = RGL }
        OR = {
            is_ally_with = GLD
            is_ally_with = VAL
            GLD = { all_core_state = { is_controlled_by = RGL } }
            VAL = { all_core_state = { is_controlled_by = RGL } }
        }
    }
    else = {
        always = no
    }
}



gob_has_overthrow_timer = {
    hidden_trigger = {
        NOT = { has_completed_focus = GOB_crack_down_on_the_clans }   
    }
    has_defensive_war = no
    check_variable = {
        var = num_of_civilian_factories
        value = factories_demand
        compare = less_than_or_equals
        tooltip = gob_satisfy_clans_tt
    }
}

gob_has_not_overthrow_timer = {
    OR = {
        check_variable = {
            var = num_of_civilian_factories
            value = factories_demand
            compare = less_than_or_equals
            tooltip = gob_satisfy_clans_tt
        }
        has_defensive_war = yes
        has_completed_focus = GOB_crack_down_on_the_clans
    }
}




# Starting a war between PREV and THIS is possible (no NA pacts, allies or existing wars)
war_with_is_possible = {
    exists = yes
    NOT = { is_ally_with = PREV }
    NOT = { has_war_with = PREV }
    NOT = { has_non_aggression_pact_with = PREV }
}
