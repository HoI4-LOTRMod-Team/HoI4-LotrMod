# Either one of the other states is an ally, or we control all their core states
is_vales_confederation_member = {
    if = {
        limit = { tag = VAL }
        OR = {
            is_ally_with = RGL
            is_ally_with = GLD
            RGL = { all_core_state = { is_controlled_by = VAL } }
            GLD = { all_core_state = { is_controlled_by = VAL } }
        }
    }
    else_if = {
        limit = { tag = GLD }
        OR = {
            is_ally_with = RGL
            is_ally_with = VAL
            RGL = { all_core_state = { is_controlled_by = GLD } }
            VAL = { all_core_state = { is_controlled_by = GLD } }
        }
    }
    else_if = {
        limit = { tag = RGL }
        OR = {
            is_ally_with = GLD
            is_ally_with = VAL
            GLD = { all_core_state = { is_controlled_by = RGL } }
            VAL = { all_core_state = { is_controlled_by = RGL } }
        }
    }
    else = {
        always = no
    }
}
