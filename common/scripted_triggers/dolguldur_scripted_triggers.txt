


any_schedule_cancelled = {
	OR = {
		hidden_trigger = {
			NOT = { has_country_flag = dgu_saurons_schedule_active }
		}
		MOR = {
			OR = {
				is_subject = yes
				exists = no
				has_capitulated = yes
				strength_ratio = {
					tag = DGU
					ratio < 0.5
				}
			}
		}
	}
}

		}
	}
}
