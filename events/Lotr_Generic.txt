# Claim state with chance of working out a deal
country_event = {
	id = lotr_generic.1
	title = lotr_generic.1.t
	desc = lotr_generic.1.d
	picture = GFX_report_event_dagger_map_war

	is_triggered_only = yes

	# Outrage response / do nothing about it
	option = {
		name = lotr_generic.1.a

		ai_chance = {
			base = 10
			modifier = {
				add = 10
				strength_ratio = {
					tag = FROM
					ratio > 1
				}
			}
		}

		event_target:target_claimed_state = {
			add_claim_by = FROM
		}
	}
	# Try to work out a deal where we get to keep the territory
	option = {
		name = lotr_generic.1.b

		ai_chance = {
			base = 10
			modifier = {
				add = 10
				strength_ratio = {
					tag = FROM
					ratio < 1.5
				}
			}
		}

		event_target:target_claimed_state = {
			add_claim_by = FROM
		}
		custom_effect_tooltip = lotr_generic.1.b_tooltip
		hidden_effect = {
			FROM = { country_event = { id = lotr_generic.2 days = 1 random_days = 1 } }
		}
	}
	# Hand it over for a non-agression pact
	option = {
		name = lotr_generic.1.c

		ai_chance = {
			base = 4
			modifier = {
				factor = 0
				strength_ratio = {
					tag = FROM
					ratio > 1
				}
			}
		}

		effect_tooltip = {

			FROM = {
				transfer_state = event_target:target_claimed_state
			}
			diplomatic_relation = {
				country = FROM
				relation = non_aggression_pact
				active = yes
			}
		}
		hidden_effect = {
			FROM = { country_event = { id = lotr_generic.5 days = 1 random_days = 1 } }
		}
	}
}

    desc = lotr_generic.125.desc

    is_triggered_only = yes

	# send supplies
    option = {
        ai_chance = {
            base = 10
			modifier = {
				factor = 2
				has_equipment = { infantry_equipment > 2000 }
			}
			modifier = {
				factor = 2
				is_ally_with = FROM
			}

				factor = 1.5
				has_war = no
			}
			modifier = {
				factor = 0.3

			}
			modifier = {
				factor = 0
				OR = {
					has_equipment = { infantry_equipment < 1200 }
					has_equipment = { support_equipment < 500 }
				}
			}
        }
        name = lotr_generic.125.a
		FROM = {
			hidden_effect = {
				country_event = { id = lotr_generic.126 days = 3 }
			}
			effect_tooltip = {
				send_equipment = {
					target = FROM
					equipment = infantry_equipment
					amount = 500
				}
				if = {
					limit = { has_equipment = { support_equipment > 100 } }
					send_equipment = {
						target = FROM
						equipment = support_equipment
						amount = 100
					}
				}
			}
		}
	}

	# send military advice and thoughts+prayers
    option = {
        ai_chance = {
            base = 10
			modifier = {
				factor = 0.3
				FROM = { we_dont_like_them = yes }
			}
        }
        name = lotr_generic.125.b
		FROM = {
			hidden_effect = {
				country_event = { id = lotr_generic.127 days = 3 }
			}
			event_option_tooltip = lotr_generic.126.b
		}
	}

    option = {
        ai_chance = {
            base = 10
			modifier = {
				factor = 2.5
				FROM = { we_dont_like_them = yes }
			}
			modifier = {
				factor = 0.1
				FROM = { we_like_them = yes }
				OR = {
					FROM = { surrender_progress > 0.1 }
					FROM = { enemies_strength_ratio > 1.0 }
				}
			}
        }
        name = lotr_generic.125.c
		add_political_power = -25
		FROM = { decrease_their_opinion_of_us_small = yes }
	}
