country_event = {
    id = spiders.123
    title = spiders.123.t
    desc = spiders.123.desc

    trigger = {
        can_get_spider_incidents = yes
    }

    mean_time_to_happen = {
        months = 6
    }

    # It can't be helped. It's their nature
    option = {
        ai_chance = {
            base = 1
        }
        name = spiders.123.a
        
        spider_incident_effect = yes
        if = {
            limit = { is_orcs = no has_government = belligerent }
            add_stability = -0.02
        }
        else_if = {
            limit = { is_orcs = no }
            add_stability = -0.04
        }
    }

    # Demmand immediate recompensation!
    option = {
        ai_chance = {
            base = 1
        }
        trigger = {
            NOT = { is_subject_of = SPI }
        }
        name = spiders.123.b
        
        spider_incident_effect = yes
        SPI = {
            decrease_their_opinion_of_us_medium = yes
        }
    }
}
	
# Spiders and elves settle territorial disputes (TODO: localization)
country_event = {
	id = spiders.124
	title = spiders.124.t
	desc = spiders.124.d
	picture = GFX_report_event_elven_alliance # TODO

	is_triggered_only = yes

	option = {
		name = spiders.124.a
		ai_chance = {
			base = 1
		}
        if = {
            limit = {
                OR = {
                    AND = { SPI = { is_ai = yes } MIR = { is_ai = no } tag = MIR }
                    AND = { SPI = { is_ai = no } tag = SPI }
                    AND = { MIR = { is_ai = yes } tag = SPI }
                }
            }
            SPI = {
                every_controlled_state = {
                    if = { limit = { is_claimed_by = MIR } remove_claim_by = MIR }
                    if = { limit = { is_core_of = MIR } remove_core_of = MIR }
                }
            }
            MIR = {
                every_controlled_state = {
                    if = { limit = { is_claimed_by = SPI } remove_claim_by = SPI }
                    if = { limit = { is_core_of = SPI } remove_core_of = SPI }
                }
            }
        }
        else = {
            effect_tooltip = {
                SPI = {
                    every_controlled_state = {
                        if = { limit = { is_claimed_by = MIR } remove_claim_by = MIR }
                        if = { limit = { is_core_of = MIR } remove_core_of = MIR }
                    }
                }
                MIR = {
                    every_controlled_state = {
                        if = { limit = { is_claimed_by = SPI } remove_claim_by = SPI }
                        if = { limit = { is_core_of = SPI } remove_core_of = SPI }
                    }
                }
            }
        }
	}
}
