		name = rhudaur.3.a

			base = 25
			modifier = {

			}
			modifier = {

			}
			modifier = {
				add = 5

				}
			}

			}

			puppet = RHD
		}
	}

	option = { #no
		name = rhudaur.3.b
		ai_chance = {
			base = 5

				factor = 999
				has_war = yes
			}
			modifier = {
				factor = 999

			}			
			modifier = {
				add = 999
				has_war_with = ANG
			}
		}
		add_war_support = 0.10
		add_opinion_modifier = { target = ANG modifier = medium_decrease }
		ANG = {
			add_opinion_modifier = { target = RHD modifier = medium_decrease }
			create_wargoal = {
				target = RHD
				type = puppet_wargoal_focus
			}
