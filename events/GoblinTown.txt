# Goblin contract expired
country_event = {
    id = goblintown.109
	picture = GFX_report_event_mercenary_offer
    title = goblintown.109.t
    desc = goblintown.109.desc

    is_triggered_only = yes

    # NOTE: I've literally made a graph plotting out the chances of the AI accepting this offer :skull:. You can see it here: https://i.imgur.com/brqPLsD.png

    # extend
    option = {
        ai_chance = {
            base = 1
            modifier = {
                set_temp_variable = { temp_str_inv = 1 }
                divide_temp_variable = { temp_str_inv = alliance_strength_ratio }
                factor = temp_str_inv
            }
            modifier = {
                factor = 2
                ai_has_under_25p_wanted_divs = yes
                alliance_strength_ratio < 1.1
            }
            modifier = {
                factor = 1.5
                ai_has_under_50p_wanted_divs = yes
                alliance_strength_ratio < 1.1
            }
            modifier = {
                # evenly matched -> need every advantage
                factor = 2
                alliance_strength_ratio > 0.75
                alliance_strength_ratio < 1.1
            }
            modifier = {
                # offer made by player -> more likely to accept
                factor = 1.3
                GOB = { is_ai = no }
            }
            modifier = {
                # we have overwhelming force -> no need
                factor = 0.001
                alliance_strength_ratio > 2.0
            }
            modifier = {
                # we are at peace -> no need
                factor = 0.001
                has_war = no
            }
            modifier = {
                 # 30% chance to refuse regardless
                factor = 0.001
                has_country_flag = rng_30p_flag_2
            }
            modifier = {
                # elves don't like goblins much
                factor = 0.25
                is_elves = yes
                alliance_strength_ratio > 0.5 # I'll take those odds!
            }
            modifier = {
                factor = 1.25
                has_government = belligerent
            }
        }
        trigger = {
            # no point in extending if they no longer have mercs
            any_country_division = {
                division_has_majority_template = goblin_mercenary_infantry
            }
        }
        name = goblintown.109.a
        GOB = {
            country_event = goblintown.110
        }
        add_timed_idea = {
            idea = GOB_mercenary_contract
            days = 180
        }
    }

    # let expire
    option = {
        ai_chance = {
            base = 1
            modifier = {
                # we are facing tough odds -> need mercs
                set_temp_variable = { temp_str = alliance_strength_ratio }
                multiply_temp_variable = { temp_str = alliance_strength_ratio } # alliance ratio squared
                factor = temp_str
            }
            modifier = {
                # we are loosing -> need mercs
                factor = 0.001
                surrender_progress > 0.12
            }
            modifier = {
                # we are outnumbered -> need mercs
                factor = 0.001
                alliance_strength_ratio < 0.5
            }
        }
        name = goblintown.109.b
        custom_effect_tooltip = goblintown.109.b_tt
        hidden_effect = {
            gob_remove_mercs_and_template = yes
            GOB = {
                gob_create_merc_divisions_equipped = yes
                country_event = goblintown.111
            }
        }
	}
}
