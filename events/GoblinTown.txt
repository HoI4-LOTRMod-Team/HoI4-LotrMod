    # extend
    option = {
        ai_chance = {
            base = 1

            # no point in extending if they no longer have mercs

    option = {
        ai_chance = {
            base = 1
        }

            GOB = {
                gob_create_merc_divisions_equipped = yes
                country_event = goblintown.111
            }
        }
	}
}

# GOB raid on RIV successful
country_event = {
    id = goblintown.119
    title = goblintown.119.t
    desc = goblintown.119.desc

    is_triggered_only = yes

    option = {
        ai_chance = {
            base = 1
        }
        trigger = {
            any_country = {
                is_neighbor_of = GOB
                OR = {
                    is_ally_with = RIV
                    tag = RIV
                }
            }
        }
        name = goblintown.119.a
		hidden_effect = {
            GOB = { country_event = { id = goblintown.121 days = 1 } }
        }
        create_wargoal = {
            target = GOB
            type = annex_everything
        }
        add_opinion_modifier = {
            target = GOB
            modifier = large_decrease
        }
    }

    option = {
        ai_chance = {
            base = 1
        }
        name = goblintown.119.b
	    hidden_effect = {
            GOB = { country_event = { id = goblintown.122 days = 3 } }
        }
        send_embargo = GOB
        add_opinion_modifier = {
            target = GOB
            modifier = large_decrease
        }
    }
}
