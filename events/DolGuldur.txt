# POV: DGU. Ask one of the kingdoms of the Vales to join a secret deal
country_event = {
	id = dolguldur.102
	title = dolguldur.102.t
	desc = dolguldur.102.desc

	is_triggered_only = yes

	option = {
		trigger = {
			RGL = {
				is_subject = no
				NOT = { has_war_with = DGU }
				is_in_faction = no
			}
		}
		ai_chance = {
			base = 1
		}
		name = dolguldur.102.a
		RGL = { country_event = { id = dolguldur.105 days = 3 } }
		custom_effect_tooltip = if_they_accept_newline_tt
		event_option_tooltip = dolguldur.108.a
	}

	option = {
		trigger = {
			GLD = {
				is_subject = no
				NOT = { has_war_with = DGU }
				is_in_faction = no
			}
		}
		ai_chance = {
			base = 1
		}
		name = dolguldur.102.b
		GLD = { country_event = { id = dolguldur.106 days = 3 } }
		custom_effect_tooltip = if_they_accept_newline_tt
		event_option_tooltip = dolguldur.110.a
	}

	option = {
		trigger = {
			VAL = {
				is_subject = no
				NOT = { has_war_with = DGU }
				is_in_faction = no
			}
		}
		ai_chance = {
			base = 1
		}
		name = dolguldur.102.c
		VAL = { country_event = { id = dolguldur.107 days = 3 } }
		custom_effect_tooltip = if_they_accept_newline_tt
		event_option_tooltip = dolguldur.112.a
	}

	# Shows up when the other options are not available
	option = {
		trigger = {
			RGL = { OR = { is_subject = yes has_war_with = DGU is_in_faction = yes } }
			GLD = { OR = { is_subject = yes has_war_with = DGU is_in_faction = yes } }
			VAL = { OR = { is_subject = yes has_war_with = DGU is_in_faction = yes } }
		}
		ai_chance = {
			base = 1
		}
		name = dolguldur.102.d

		add_political_power = 150
	}
}

# Rhosgobel called to the secret deal
country_event = {
	id = dolguldur.114
	title = dolguldur.114.t
	desc = dolguldur.114.desc

	fire_only_once = yes
	trigger = {
		tag = RGL
		has_country_flag = DGU_secret_deal_accepted
		NOT = { has_war_with = DGU }
		is_subject = no
		
		DGU = { has_war_with = GLD }
	}

	option = {
		ai_chance = {
			base = 1
		}
		name = dolguldur.114.a

		declare_war_on = {
			target = GLD
			type = annex_everything
		}
		give_military_access = DGU
	}

	option = {
		trigger = {
			has_war = yes
		}
		ai_chance = {
			base = 1
		}
		name = dolguldur.114.b

		add_political_power = -500
		add_stability = -0.15
	}
}
