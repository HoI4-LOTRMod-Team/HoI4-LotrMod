# Gundabad expedition asks for help
country_event = {
	id = gundabad.4
	title = gundabad.4.t
	desc = gundabad.4.d

	picture = GFX_report_event_orcs

	is_triggered_only = yes

	# Send a force
	option = {
		ai_chance = {
			base = 5
			modifier = {
				add = 50
				has_country_flag = gameplan_going_cooperative
			}
			modifier = {
				factor = 25
				has_country_flag = gameplan_going_unaligned
			}
		}
		name = gundabad.4.a

		if = {
			limit = { tag = ERE }
			add_manpower = -1250
			add_equipment_to_stockpile = { type = infantry_equipment amount = -350 }
		}

		GUN = {
			add_opinion_modifier = {
				target = ROOT
				modifier = large_increase
			}
			hidden_effect = {
				country_event = gundabad.5
			}
		}
	}

	# We don't need to interfere
	option = {
		ai_chance = {
			base = 1
			modifier = {
				add = 999
				is_orcs = yes
			}
			modifier = {
				add = 5
				has_war = yes
			}
			modifier = {
				add = 15
				has_country_flag = gameplan_going_belligerent
			}
			modifier = {
				add = 10
				has_country_flag = gameplan_going_revolutionary
			}
		}
		name = gundabad.4.b

		set_country_flag = gundabad_aid_denied # Gundabad will remember this

		custom_effect_tooltip = gundabad.4.b_tt

		add_political_power = -75

		GUN = {
			add_opinion_modifier = {
				target = THIS
				modifier = small_decrease
			}
			hidden_effect = {
				country_event = gundabad.6
			}
		}
	}
}

}
