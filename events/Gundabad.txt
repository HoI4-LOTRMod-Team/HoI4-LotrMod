# Auto-complete Expedition focus
country_event = {
    id = gundabad.105
	picture = GFX_report_event_orcs_2
    title = gundabad.105.t
    desc = gundabad.105.desc

    trigger = {
		tag = GUN
		has_completed_focus = GUN_the_gundabad_expedition
		NOT = { has_completed_focus = GUN_the_new_inhabitants }
		power_balance_value = {
			id = power_balance_gundabad
			value > 0.99
		}
	}

	mean_time_to_happen = {
		days = 70
	}

    option = {
        ai_chance = {
            base = 1
        }
        name = gundabad.105.a

		if = {
			limit = {
				NOT = { has_completed_focus = GUN_rally_the_scattered_longbeards }
				NOT = { has_completed_focus = GUN_gather_eotheod_descendants }
			}
			random_list = {
				50 = {
					complete_national_focus = GUN_rally_the_scattered_longbeards
				}
				50 = {
					complete_national_focus = GUN_gather_eotheod_descendants
				}
			}	
		}

		complete_national_focus = GUN_the_new_inhabitants
    }
}
