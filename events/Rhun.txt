# Treaty of the Steppe
country_event = {
    id = rhun.116
	picture = GFX_report_event_intense_diplomatic_negotiations
    title = rhun.116.t
    desc = rhun.116.desc

	fire_only_once = yes

	trigger = {
		tag = DRH
		has_civil_war = yes
		country_exists = RHU
		surrender_progress > 0.7
	}

	mean_time_to_happen = {
		days = 1
	}

    option = {
        ai_chance = {
            base = 50
        }
        name = rhun.116.a
		complete_national_focus = DRH_treaty_of_the_steppe
		RHU = {
			add_political_power = 150
			add_war_support = 0.1
		}
		DRH = {
			add_political_power = -150
			add_war_support = -0.1
		}
		custom_effect_tooltip = rhun.116.a
		hidden_effect = {
			DRH = {
				every_controlled_state = {
					set_state_flag = treaty_of_the_steppe_drh_control
				}
			}
			RHU = {
				every_controlled_state = {
					set_state_flag = treaty_of_the_steppe_rhun_control
				}
			}
			DRH = {
				white_peace = RHU
				set_truce = {
					target = RHU
					days = 365
				}
			}
			every_state = {
				limit = { has_state_flag = treaty_of_the_steppe_drh_control }
				transfer_state_to = DRH
			}
			every_state = {
				limit = { has_state_flag = treaty_of_the_steppe_rhun_control }
				transfer_state_to = RHU
			}
		}
		128 = {
			transfer_state_to = DRH
		}
    }

    option = {
        ai_chance = {
            base = 50
        }
        name = rhun.116.b
		add_war_support = 0.1
    }
}
